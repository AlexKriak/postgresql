CREATE OR REPLACE FUNCTION create_delivery_after_order_packing()
RETURNS TRIGGER AS $$
DECLARE
    order_id INTEGER;
BEGIN
    order_id := NEW.id;

    IF NEW.status = 'packing' AND OLD.status <> 'packing' THEN
        PERFORM 1 FROM inventory.deliveries WHERE order_id = order_id LIMIT 1;
        IF NOT FOUND THEN
            INSERT INTO inventory.deliveries (order_id, status, created_at)
            VALUES (order_id, 'pending', NOW());
            RAISE NOTICE 'Delivery created for order ID %.', order_id;
        ELSE
            RAISE NOTICE 'Delivery already exists for order ID %.', order_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_create_delivery_after_order_update
AFTER UPDATE OF status ON sales.orders
FOR EACH ROW
WHEN (OLD.status IS DISTINCT FROM NEW.status AND NEW.status = 'packing')
EXECUTE FUNCTION create_delivery_after_order_packing();
