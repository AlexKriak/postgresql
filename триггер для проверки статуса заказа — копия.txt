CREATE OR REPLACE FUNCTION check_order_fulfillment_and_update_status()
RETURNS TRIGGER AS $$
DECLARE
    order_id INTEGER;
    all_items_reserved BOOLEAN := FALSE;
    order_status TEXT;
BEGIN
    order_id := NEW.order_id;
    SELECT status INTO order_status FROM sales.orders WHERE id = order_id;

    IF order_status = 'processing' THEN
        SELECT NOT EXISTS (
            SELECT 1
            FROM inventory.reserves ir
            JOIN sales.order_items oi ON ir.order_id = oi.orders_id AND ir.product_id = oi.product_id
            WHERE ir.order_id = order_id
              AND ir.status NOT IN ('fulfilled', 'pending_transfer')
        ) INTO all_items_reserved;

        IF all_items_reserved THEN
            UPDATE sales.orders
            SET status = 'packing'
            WHERE id = order_id;
            RAISE NOTICE 'Order ID % status updated to packing after all items reserved.', order_id;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_check_order_status_after_reserve_update
AFTER UPDATE OF status ON inventory.reserves
FOR EACH ROW
WHEN (OLD.status IS DISTINCT FROM NEW.status)
EXECUTE FUNCTION check_order_fulfillment_and_update_status();
